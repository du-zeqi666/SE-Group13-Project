\begin{frame}{三种距离度量方式与项目统计}

\textbf{系统支持三种距离度量，构建索引时选定，搜索时锁定。}

\vspace{0.3em}
\ThreeLineTable{%
  \begin{tabular}{p{2.5cm} p{4cm} p{4cm} p{3cm}}
    \toprule
    \textbf{度量方式} & \textbf{数学含义} & \textbf{实现细节} & \textbf{选用建议} \\
    \midrule
    L2 (Euclidean)
    & $\sqrt{\sum_i(A_i-B_i)^2}$，空间直线距离
    & FAISS IndexFlatL2，Annoy euclidean
    & 数值归一化后 \\
    \hline
    Cosine
    & $\frac{A\cdot B}{|A||B|}$，方向夹角
    & L2归一化 + FAISS IndexFlatIP，Annoy angular
    & $\checkmark$ \textbf{推荐}：不受表达量影响 \\
    \hline
    Inner Product
    & $\sum_i A_i B_i$，点积大小
    & FAISS IndexFlatIP / METRIC\_INNER\_PRODUCT，Annoy dot
    & 有语义的内积场景 \\
    \bottomrule
  \end{tabular}
}

\vspace{0.3em}
\begin{columns}[T]
  \begin{column}{0.50\textwidth}
    \begin{block}{为什么 Cosine 是单细胞分析的默认选择？}
      基因表达量受测序深度影响大，余弦相似度关心\textbf{基因相对表达模式的方向相似性}，不受总量影响。
      L2 归一化后 Cosine $\equiv$ Inner Product。
    \end{block}
  \end{column}
  \begin{column}{0.46\textwidth}
    \centering
    \footnotesize
    \ThreeLineTable{%

\begin{frame}{RAG 自然语言检索与联合索引}

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{RAG 智能检索流程}
    \begin{enumerate}
      \item 用户输入自然语言（如"找与肝癌细胞相似的 T 细胞"）
      \item Sentence-Transformers 编码查询语义
      \item 自动提取元数据过滤条件（cell\_type、disease 等）
      \item ChromaDB HNSW 语义相似搜索
      \item AI 对 Top-K 结果统计分析并生成自然语言解读
    \end{enumerate}

    \vspace{0.3em}
    \textbf{联合索引（Joint Index）}
    \begin{itemize}
      \item ChromaDB 多数据集合并索引，支持\textbf{跨数据集检索}
      \item 独立管理界面，含构建、查询、列表、删除功能
    \end{itemize}
