\begin{frame}[fragile]{核心算法逻辑：小批量梯度下降}
  % 注意：[H] 参数极其重要，Beamer 不支持 algorithm 的自动浮动机制
  \begin{algorithm}[H]
    \caption{Mini-Batch Gradient Descent}
    \SetAlgoLined
    
    \KwIn{原始数据集 $\mathcal{D}$, 学习率 $\eta$, 迭代次数 $T$, 批次大小 $B$}
    \KwOut{优化后的模型权重 $W_T$}
    
    初始化权重 $W_0 \sim \mathcal{N}(0, \sigma^2)$\;
    
    \For{$t = 1$ \KwTo $T$}{
      从 $\mathcal{D}$ 中均匀无放回采样 $B$ 个样本，构成小批量 $\mathcal{B}_t$\;
      计算该批次的平均经验损失：
      $L_{\mathcal{B}_t} = \frac{1}{B} \sum_{x_i \in \mathcal{B}_t} \ell(f(x_i; W_{t-1}), y_i)$\;
      反向传播，计算梯度：$g_t = \nabla_{W} L_{\mathcal{B}_t}$\;
      更新模型权重：$W_t \leftarrow W_{t-1} - \eta g_t$\;
    }
    \Return $W_T$\;
  \end{algorithm}
\end{frame}

\begin{frame}{年度综合绩效评估模型}
  核心公式解析：

  \vspace{1em}

  \begin{center}
    \BehindEqBox{
      $\displaystyle
      S = \underbrace{\sum_{i=1}^{N} \textcolor{myorange}{w_i}\, \textcolor{mygreen}{s_i}}_{\textcolor{midpurple}{\textbf{基础得分}}} \;+\;
      \overbrace{\left( {\frac{1}{N} \sum_{j=1}^{M} \textcolor{myred}{\delta_j}} \right)}^{\textcolor{mypurple}{\textbf{环境调节因子}}}
      $
    }
  \end{center}

  \vspace{1em}

  % 解释公式变量
  \[
    \left\{
    \begin{aligned}
      \textcolor{mygreen}{s_i} &\;=\; \text{模块 } i \text{ 的标准化评分}, \\
      \textcolor{myorange}{w_i} &\;=\; \text{模块 } i \text{ 的权重系数}, \\
      \textcolor{myred}{\delta_j} &\;=\; \text{外部因素 } j \text{ 带来的修正值}.
    \end{aligned}
    \right.
  \]

  % \vspace{0.5em}

  \textbf{说明：} 此处 \(S\) 代表年度综合评估指标。其中 \textcolor{midpurple}{基础得分} 是各模块标准化评分 \(s_i\) 的加权总和，而 \textcolor{mypurple}{环境调节因子} 则用于衡量外部影响的平均修正。

\end{frame}
